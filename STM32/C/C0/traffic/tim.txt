with System;
with STM32_SVD.TIM; -- Adjust to your specific SVD package name

package body Hardware.TIM1 is

   procedure Init_TIM1_PWM is
   begin
      -- 1. Configure Channel Modes (CCMR1 for Ch1/Ch2, CCMR2 for Ch4)
      --    Set Output Compare 1/2/4 to PWM Mode 1 (110) with Preload Enable (OCxPE)

      TIM1_Periph.CCMR1_Output.OC1M  := 2#110#; -- PWM Mode 1
      TIM1_Periph.CCMR1_Output.OC1PE := True;   -- Preload enable

      TIM1_Periph.CCMR2_Output.OC2M  := 2#110#; -- PWM Mode 1
      TIM1_Periph.CCMR2_Output.OC2PE := True;   -- Preload enable

      TIM1_Periph.CCMR2_Output.OC4M  := 2#110#; -- PWM Mode 1
      TIM1_Periph.CCMR2_Output.OC4PE := True;   -- Preload enable

      -- 2. Enable Outputs in Channel Capture/Compare Enable Register (CCER)
      TIM1_Periph.CCER.CC1E := True; -- Enable Channel 1 output
      TIM1_Periph.CCER.CC2E := True; -- Enable Channel 2 output
      TIM1_Periph.CCER.CC4E := True; -- Enable Channel 4 output

      -- 3. Set Frequency / Resolution (PSC and ARR)
      --    Example: For a 10-bit PWM resolution (0..1023), set ARR = 1023
      TIM1_Periph.PSC := 0;      -- Set prescaler as needed for clock speed
      TIM1_Periph.ARR := 1023;   -- Period match value

      -- 4. Enable Auto-Reload Preload (CR1.ARPE)
      TIM1_Periph.CR1.ARPE := True;

      -- 5. Set Initial Compare Values (Duty Cycles to 0)
      TIM1_Periph.CCR1 := 0;
      TIM1_Periph.CCR2 := 0;
      TIM1_Periph.CCR4 := 0;

      -- 6. TIM1 ADVANCED TIMER MUST-HAVES:
      --    Enable Main Output Enable (MOE) in BDTR
      TIM1_Periph.BDTR.MOE := True;

      --    Trigger an Update Event (UG) to latch PSC, ARR, and CCMR shadow registers
      TIM1_Periph.EGR.UG := True;

      -- 7. Enable Counter
      TIM1_Periph.CR1.CEN := True;
   end Init_TIM1_PWM;

end Hardware.TIM1;
Crucial Register Gotchas for Channels 1, 2, and 4
CCMR Mapping Difference:

CCMR1 manages Channel 1 and Channel 2.

CCMR2 manages Channel 3 and Channel 4. Ensure you set OC4M in CCMR2_Output for Channel 4 (rather than looking for it in CCMR1).

Channel 3 Disabled:
Leave CCER.CC3E as False and CCMR2.OC3M as 0 if Channel 3 is unused, so its GPIO pin remains free for other functions.

BDTR.MOE Requirement:
Because TIM1 is an advanced timer, skipping TIM1_Periph.BDTR.MOE := True will leave the timer counter running internally, but no PWM signals will appear on the GPIO pins for Channels 1, 2, or 4.
